cmake_minimum_required(VERSION 3.24)

project(worldfoundry_native_kernels VERSION 0.1.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

find_package(Python3 COMPONENTS Interpreter REQUIRED)

execute_process(
  COMMAND
    "${Python3_EXECUTABLE}" -c
    "import torch; from torch.utils.cpp_extension import include_paths; print(';'.join(include_paths()))"
  RESULT_VARIABLE WORLDFOUNDRY_TORCH_PROBE_RESULT
  OUTPUT_VARIABLE WORLDFOUNDRY_TORCH_INCLUDE_DIRS
  ERROR_VARIABLE WORLDFOUNDRY_TORCH_PROBE_ERROR
  OUTPUT_STRIP_TRAILING_WHITESPACE
)
if(NOT WORLDFOUNDRY_TORCH_PROBE_RESULT EQUAL 0)
  message(FATAL_ERROR
    "The native wheel must be built in an environment with the target PyTorch already installed. "
    "Use --no-build-isolation. Probe error: ${WORLDFOUNDRY_TORCH_PROBE_ERROR}"
  )
endif()

execute_process(
  COMMAND
    "${Python3_EXECUTABLE}" -c
    "import torch; from torch.utils.cpp_extension import library_paths; print(';'.join(library_paths()))"
  OUTPUT_VARIABLE WORLDFOUNDRY_TORCH_LIBRARY_DIRS
  OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY
)
unset(WORLDFOUNDRY_TORCH_CPU_LIBRARY CACHE)
unset(WORLDFOUNDRY_C10_LIBRARY CACHE)
find_library(WORLDFOUNDRY_TORCH_CPU_LIBRARY torch_cpu
  PATHS ${WORLDFOUNDRY_TORCH_LIBRARY_DIRS}
  NO_DEFAULT_PATH
  NO_CACHE
  REQUIRED
)
find_library(WORLDFOUNDRY_C10_LIBRARY c10
  PATHS ${WORLDFOUNDRY_TORCH_LIBRARY_DIRS}
  NO_DEFAULT_PATH
  NO_CACHE
  REQUIRED
)

execute_process(
  COMMAND "${Python3_EXECUTABLE}" -c "import torch; print(torch.__version__)"
  OUTPUT_VARIABLE WORLDFOUNDRY_TORCH_VERSION
  OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY
)
execute_process(
  COMMAND "${Python3_EXECUTABLE}" -c "import torch; print(torch.version.cuda or '')"
  OUTPUT_VARIABLE WORLDFOUNDRY_TORCH_CUDA
  OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY
)
execute_process(
  COMMAND "${Python3_EXECUTABLE}" -c
          "import torch; f=getattr(torch, 'compiled_with_cxx11_abi', None); print(int(f() if f else torch._C._GLIBCXX_USE_CXX11_ABI))"
  OUTPUT_VARIABLE WORLDFOUNDRY_CXX11_ABI
  OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY
)
execute_process(
  COMMAND "${CMAKE_CXX_COMPILER}" -print-file-name=libstdc++.so.6
  OUTPUT_VARIABLE WORLDFOUNDRY_LIBSTDCXX_PATH
  OUTPUT_STRIP_TRAILING_WHITESPACE
  COMMAND_ERROR_IS_FATAL ANY
)
if(EXISTS "${WORLDFOUNDRY_LIBSTDCXX_PATH}")
  get_filename_component(WORLDFOUNDRY_LIBSTDCXX_PATH "${WORLDFOUNDRY_LIBSTDCXX_PATH}" REALPATH)
  file(SHA256 "${WORLDFOUNDRY_LIBSTDCXX_PATH}" WORLDFOUNDRY_LIBSTDCXX_SHA256)
else()
  set(WORLDFOUNDRY_LIBSTDCXX_PATH "unknown")
  set(WORLDFOUNDRY_LIBSTDCXX_SHA256 "unknown")
endif()

set(WORLDFOUNDRY_OPERATOR_ABI_VERSION 1)
set(WORLDFOUNDRY_MANIFEST_SCHEMA_VERSION 1)
set(WORLDFOUNDRY_SOURCE_REVISION "unknown" CACHE STRING "WorldFoundry source revision recorded in the wheel")

set(WORLDFOUNDRY_NATIVE_SOURCE_FILES
  CMakeLists.txt
  pyproject.toml
  cmake/write_build_manifest.cmake
  csrc/registration.cpp
  csrc/build_info.cpp
  python/worldfoundry_native_kernels/__init__.py
  python/worldfoundry_native_kernels/_loader.py
)
set(WORLDFOUNDRY_SOURCE_HASH_INPUT "")
foreach(WORLDFOUNDRY_SOURCE_FILE IN LISTS WORLDFOUNDRY_NATIVE_SOURCE_FILES)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/${WORLDFOUNDRY_SOURCE_FILE}" WORLDFOUNDRY_SOURCE_FILE_HASH)
  string(APPEND WORLDFOUNDRY_SOURCE_HASH_INPUT
    "${WORLDFOUNDRY_SOURCE_FILE}:${WORLDFOUNDRY_SOURCE_FILE_HASH}|"
  )
endforeach()
string(SHA256 WORLDFOUNDRY_SOURCE_TREE_HASH "${WORLDFOUNDRY_SOURCE_HASH_INPUT}")
string(SHA256 WORLDFOUNDRY_OPERATOR_SCHEMA_HASH
  "worldfoundry_native::_build_info() -> str|abi=${WORLDFOUNDRY_OPERATOR_ABI_VERSION}"
)
set(WORLDFOUNDRY_BUILD_ID_INPUT
  "${PROJECT_VERSION}|${WORLDFOUNDRY_OPERATOR_SCHEMA_HASH}|${WORLDFOUNDRY_SOURCE_TREE_HASH}|${WORLDFOUNDRY_TORCH_VERSION}|${WORLDFOUNDRY_TORCH_CUDA}|${WORLDFOUNDRY_CXX11_ABI}|${CMAKE_CXX_COMPILER_ID}|${CMAKE_CXX_COMPILER_VERSION}"
)
string(SHA256 WORLDFOUNDRY_BUILD_ID_FULL "${WORLDFOUNDRY_BUILD_ID_INPUT}")
string(SUBSTRING "${WORLDFOUNDRY_BUILD_ID_FULL}" 0 24 WORLDFOUNDRY_BUILD_ID)

add_library(worldfoundry_native_schema SHARED
  csrc/registration.cpp
  csrc/build_info.cpp
)
set_target_properties(worldfoundry_native_schema PROPERTIES
  BUILD_WITH_INSTALL_RPATH YES
  CXX_VISIBILITY_PRESET hidden
  INSTALL_RPATH ""
  OUTPUT_NAME "_worldfoundry_native_schema"
  PREFIX ""
  VISIBILITY_INLINES_HIDDEN YES
)
target_include_directories(worldfoundry_native_schema PRIVATE ${WORLDFOUNDRY_TORCH_INCLUDE_DIRS})
target_link_libraries(worldfoundry_native_schema PRIVATE
  "${WORLDFOUNDRY_TORCH_CPU_LIBRARY}"
  "${WORLDFOUNDRY_C10_LIBRARY}"
)
target_compile_definitions(worldfoundry_native_schema PRIVATE
  _GLIBCXX_USE_CXX11_ABI=${WORLDFOUNDRY_CXX11_ABI}
  WORLDFOUNDRY_NATIVE_BUILD_ID="${WORLDFOUNDRY_BUILD_ID}"
  WORLDFOUNDRY_NATIVE_OPERATOR_ABI_VERSION=${WORLDFOUNDRY_OPERATOR_ABI_VERSION}
)

set(WORLDFOUNDRY_BUILD_MANIFEST "${CMAKE_CURRENT_BINARY_DIR}/build_manifest.json")
add_custom_command(TARGET worldfoundry_native_schema POST_BUILD
  COMMAND "${CMAKE_COMMAND}"
    "-DINPUT_LIBRARY=$<TARGET_FILE:worldfoundry_native_schema>"
    "-DOUTPUT_MANIFEST=${WORLDFOUNDRY_BUILD_MANIFEST}"
    "-DMANIFEST_SCHEMA_VERSION=${WORLDFOUNDRY_MANIFEST_SCHEMA_VERSION}"
    "-DOPERATOR_ABI_VERSION=${WORLDFOUNDRY_OPERATOR_ABI_VERSION}"
    "-DOPERATOR_SCHEMA_HASH=${WORLDFOUNDRY_OPERATOR_SCHEMA_HASH}"
    "-DBUILD_ID=${WORLDFOUNDRY_BUILD_ID}"
    "-DSOURCE_REVISION=${WORLDFOUNDRY_SOURCE_REVISION}"
    "-DSOURCE_TREE_HASH=${WORLDFOUNDRY_SOURCE_TREE_HASH}"
    "-DPACKAGE_VERSION=${PROJECT_VERSION}"
    "-DTORCH_VERSION=${WORLDFOUNDRY_TORCH_VERSION}"
    "-DTORCH_CUDA=${WORLDFOUNDRY_TORCH_CUDA}"
    "-DCXX11_ABI=${WORLDFOUNDRY_CXX11_ABI}"
    "-DCXX_COMPILER_ID=${CMAKE_CXX_COMPILER_ID}"
    "-DCXX_COMPILER_VERSION=${CMAKE_CXX_COMPILER_VERSION}"
    "-DCXX_STANDARD=${CMAKE_CXX_STANDARD}"
    "-DLIBSTDCXX_PATH=${WORLDFOUNDRY_LIBSTDCXX_PATH}"
    "-DLIBSTDCXX_SHA256=${WORLDFOUNDRY_LIBSTDCXX_SHA256}"
    -P "${CMAKE_CURRENT_SOURCE_DIR}/cmake/write_build_manifest.cmake"
  VERBATIM
)

install(TARGETS worldfoundry_native_schema
  LIBRARY DESTINATION worldfoundry_native_kernels/lib
  RUNTIME DESTINATION worldfoundry_native_kernels/lib
)
install(FILES "${WORLDFOUNDRY_BUILD_MANIFEST}"
  DESTINATION worldfoundry_native_kernels
)
